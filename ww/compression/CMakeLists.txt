# The launcher must not inherit the runtime's non-PIC objects or link options.
set(ww_decoder_targets ww_xz_decoder)
if(WW_PACK_RUNTIME)
  list(APPEND ww_decoder_targets ww_launcher_xz_decoder)
endif()
foreach(ww_decoder_target IN LISTS ww_decoder_targets)
  add_library(${ww_decoder_target} STATIC xz_decoder.c)
  target_include_directories(${ww_decoder_target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
  if(ww_decoder_target STREQUAL "ww_launcher_xz_decoder")
    target_link_libraries(${ww_decoder_target} PRIVATE XZEmbedded::Launcher)
  else()
    target_link_libraries(${ww_decoder_target} PRIVATE XZEmbedded::XZEmbedded)
  endif()
  target_compile_features(${ww_decoder_target} PUBLIC c_std_99)
endforeach()
add_library(WaterWall::XZDecoder ALIAS ww_xz_decoder)
if(TARGET waterwall_runtime_options)
  target_link_libraries(ww_xz_decoder PRIVATE waterwall_runtime_options)
endif()
