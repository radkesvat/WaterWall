cmake_minimum_required(VERSION 3.25)
project(WaterwallWindowsLauncherFixture LANGUAGES C RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P MATCHES "^(4|8)$")
  message(FATAL_ERROR "This fixture requires a Windows x86/x64 target")
endif()
get_filename_component(WATERWALL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(WW_PACK_RUNTIME ON)
set(WW_USE_STARTUP_GUARD OFF)
set(Waterwall_VERSION 1.46.9)
set(WATERWALL_APPLICATION_TARGET WaterwallApplication)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(WW_PACKED_TARGET windows-x86_64)
else()
  set(WW_PACKED_TARGET windows-x86)
endif()
include(CTest)
add_library(fixture_companion SHARED companion.c)
set_target_properties(fixture_companion PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON
  RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
add_subdirectory("${WATERWALL_SOURCE_DIR}/ww/vendor/xz-embedded" xz-embedded)
add_subdirectory("${WATERWALL_SOURCE_DIR}/ww/compression" decoder)
include("${WATERWALL_SOURCE_DIR}/tests/unittests/windows_packed_startup.cmake")
add_executable(WaterwallApplication fixture.c "${WATERWALL_SOURCE_DIR}/core/startup_options.c"
  "${WATERWALL_SOURCE_DIR}/ww/node_builder/config_lexical.c"
  "${WATERWALL_SOURCE_DIR}/ww/vendor/cjson/cJSON.c")
target_link_libraries(WaterwallApplication PRIVATE fixture_companion)
target_link_libraries(WaterwallApplication PRIVATE user32 iphlpapi)
target_include_directories(WaterwallApplication PRIVATE "${WATERWALL_SOURCE_DIR}/ww/devices")
target_compile_definitions(WaterwallApplication PRIVATE _WIN32_WINNT=0x0600 WW_CJSON_CRT_ALLOCATOR=1)
target_include_directories(WaterwallApplication PRIVATE "${WATERWALL_SOURCE_DIR}/core"
  "${WATERWALL_SOURCE_DIR}/ww/node_builder" "${WATERWALL_SOURCE_DIR}/ww/vendor/cjson")
set_target_properties(WaterwallApplication PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/$<CONFIG>")
if(MSVC)
  target_link_options(WaterwallApplication PRIVATE /DYNAMICBASE:NO /FIXED /INCREMENTAL:NO
    "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    target_link_options(WaterwallApplication PRIVATE /HIGHENTROPYVA:NO)
  endif()
else()
  target_compile_options(WaterwallApplication PRIVATE -fno-pic -fno-pie)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    target_link_options(WaterwallApplication PRIVATE "LINKER:--disable-high-entropy-va")
  endif()
  target_link_options(WaterwallApplication PRIVATE -static "LINKER:--disable-dynamicbase" "LINKER:--disable-reloc-section")
  add_library(waterwall_resources OBJECT "${WATERWALL_SOURCE_DIR}/core/rc/app.rc")
  target_include_directories(waterwall_resources PRIVATE "${WATERWALL_SOURCE_DIR}/core/rc")
  target_sources(WaterwallApplication PRIVATE $<TARGET_OBJECTS:waterwall_resources>)
endif()
add_subdirectory("${WATERWALL_SOURCE_DIR}/core/launcher" launcher)
target_compile_definitions(Waterwall PRIVATE WATERWALL_LAUNCHER_TEST_HOOKS=1)
find_program(PYTHON_EXECUTABLE NAMES python3 python REQUIRED)
add_test(NAME waterwall.windows_packed_fixture
  COMMAND "${PYTHON_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/smoke.py"
    --launcher "$<TARGET_FILE:Waterwall>" --application "$<TARGET_FILE:WaterwallApplication>"
    ${WW_FIXTURE_RUNNER_ARGS})
set_tests_properties(waterwall.windows_packed_fixture PROPERTIES TIMEOUT 120)
add_executable(windows_lifecycle_launcher_test lifecycle.c)
add_dependencies(windows_lifecycle_launcher_test Waterwall)
target_include_directories(windows_lifecycle_launcher_test PRIVATE "${WATERWALL_SOURCE_DIR}/core")
target_compile_definitions(windows_lifecycle_launcher_test PRIVATE _WIN32_WINNT=0x0600)
if(NOT MSVC)
  target_link_options(windows_lifecycle_launcher_test PRIVATE -static)
endif()
if(CMAKE_HOST_WIN32)
  add_test(NAME waterwall.windows_lifecycle_launcher_fixture
    COMMAND windows_lifecycle_launcher_test "$<TARGET_FILE:Waterwall>")
  set_tests_properties(waterwall.windows_lifecycle_launcher_fixture PROPERTIES TIMEOUT 150)
endif()

add_executable(windows_session_effects_test effects.c "${WATERWALL_SOURCE_DIR}/ww/devices/windows_session_effects.c")
target_include_directories(windows_session_effects_test PRIVATE "${WATERWALL_SOURCE_DIR}/ww/devices")
target_compile_definitions(windows_session_effects_test PRIVATE _WIN32_WINNT=0x0600)
target_link_libraries(windows_session_effects_test PRIVATE iphlpapi)
add_test(NAME waterwall.windows_session_effects_unit COMMAND windows_session_effects_test)
set_tests_properties(waterwall.windows_session_effects_unit PROPERTIES TIMEOUT 30)

add_executable(windows_tun_ownership_test ownership.c)
target_include_directories(windows_tun_ownership_test PRIVATE "${WATERWALL_SOURCE_DIR}/ww/devices/tun")
target_compile_definitions(windows_tun_ownership_test PRIVATE _WIN32_WINNT=0x0600)
target_link_libraries(windows_tun_ownership_test PRIVATE advapi32 bcrypt setupapi)
add_test(NAME waterwall.windows_tun_ownership_unit COMMAND windows_tun_ownership_test)
set_tests_properties(waterwall.windows_tun_ownership_unit PROPERTIES TIMEOUT 30)

add_executable(windows_tun_dns_test "${WATERWALL_SOURCE_DIR}/tests/unittests/tun_windows_dns_test.c")
target_include_directories(windows_tun_dns_test PRIVATE "${WATERWALL_SOURCE_DIR}/ww/devices/tun")
target_compile_definitions(windows_tun_dns_test PRIVATE _WIN32_WINNT=0x0600)
add_test(NAME waterwall.windows_tun_dns_unit COMMAND windows_tun_dns_test)
set_tests_properties(waterwall.windows_tun_dns_unit PROPERTIES TIMEOUT 30)
